import { GameState, Player, Enemy, Bullet, Particle } from '../types/game';

interface UseGameLoopProps {
  gameState: GameState;
  setGameState: (state: GameState) => void;
  canvasRef: React.RefObject<HTMLCanvasElement>;
}

export const useGameLoop = ({ gameState, setGameState, canvasRef }: UseGameLoopProps) => {
  const animationFrameRef = useRef<number>();
  const lastTimeRef = useRef<number>(0);

  const update = useCallback((deltaTime: number) => {
    if (gameState.isGameOver || gameState.isPaused) return;

    setGameState(prev => {
      const newState = { ...prev };
      
      // プレイヤーの移動
      if (newState.keysPressed.ArrowLeft || newState.keysPressed.a) {
        newState.player.x = Math.max(0, newState.player.x - 5);
      }
      if (newState.keysPressed.ArrowRight || newState.keysPressed.d) {
        newState.player.x = Math.min(newState.canvasWidth - newState.player.width, newState.player.x + 5);
      }

      // 敵の移動
      let shouldReverse = false;
      newState.enemies.forEach(enemy => {
        enemy.x += enemy.direction * enemy.speed;
        if (enemy.x <= 0 || enemy.x + enemy.width >= newState.canvasWidth) {
          shouldReverse = true;
        }
      });

      if (shouldReverse) {
        newState.enemies.forEach(enemy => {
          enemy.direction *= -1;
          enemy.y += 20;
        });
      }

      // 敵の発射
      if (Math.random() < 0.02) {
        const shooter = newState.enemies[Math.floor(Math.random() * newState.enemies.length)];
        if (shooter) {
          newState.enemyBullets.push({
            x: shooter.x + shooter.width / 2,
            y: shooter.y + shooter.height,
            width: 4,
            height: 10,
            speed: 4,
          });
        }
      }

      // 弾の移動
      newState.bullets = newState.bullets.filter(bullet => {
        bullet.y -= bullet.speed;
        return bullet.y > 0;
      });

      newState.enemyBullets = newState.enemyBullets.filter(bullet => {
        bullet.y += bullet.speed;
        return bullet.y < newState.canvasHeight;
      });

      // 衝突判定
      newState.bullets = newState.bullets.filter(bullet => {
        let hit = false;
        newState.enemies = newState.enemies.filter(enemy => {
          if (!hit && 
              bullet.x < enemy.x + enemy.width &&
              bullet.x + bullet.width > enemy.x &&
              bullet.y < enemy.y + enemy.height &&
              bullet.y + bullet.height > enemy.y) {
            hit = true;
            // エフェクト
            for (let i = 0; i < 5; i++) {
              newState.particles.push({
                x: enemy.x + enemy.width / 2,
                y: enemy.y + enemy.height / 2,
                vx: (Math.random() - 0.5) * 4,
                vy: (Math.random() - 0.5) * 4,
                life: 30,
                color: `hsl(${Math.random() * 60 + 180}, 100%, 70%)`,
              });
            }
            newState.score += 100;
            return false;
          }
          return true;
        });
        return !hit;
      });

      // 敵弾とプレイヤーの衝突
      newState.enemyBullets = newState.enemyBullets.filter(bullet => {
        if (
          bullet.x < newState.player.x + newState.player.width &&
          bullet.x + bullet.width > newState.player.x &&
          bullet.y < newState.player.y + newState.player.height &&
          bullet.y + bullet.height > newState.player.y
        ) {
          newState.lives--;
          if (newState.lives <= 0) {
            newState.isGameOver = true;
          }
          return false;
        }
        return true;
      });

      // 敵がプレイヤーに到達
      newState.enemies.forEach(enemy => {
        if (enemy.y + enemy.height >= newState.player.y) {
          newState.isGameOver = true;
        }
      });

      // パーティクル更新
      newState.particles = newState.particles.filter(p => {
        p.x += p.vx;
        p.y += p.vy;
        p.life--;
        return p.life > 0;
      });

      // 次のステージ
      if (newState.enemies.length === 0) {
        newState.level++;
        newState.enemies = Array.from({ length: 8 + newState.level * 2 }, (_, i) => ({
          x: 50 + (i % 8) * 60,
          y: 50 + Math.floor(i / 8) * 50,
          width: 40,
          height: 30,
          speed: 1 + newState.level * 0.2,
          direction: 1,
        }));
        newState.bullets = [];
        newState.enemyBullets = [];
      }

      return newState;
    });
  }, [gameState.isGameOver, gameState.isPaused]);

  const render = useCallback(() => {
    const canvas = canvasRef.current;
    if (!canvas) return;

    const ctx = canvas.getContext('2d');
    if (!ctx) return;

    // 背景
    ctx.fillStyle = '#0a0a1a';
    ctx.fillRect(0, 0, canvas.width, canvas.height);

    // 星
    ctx.fillStyle = '#ffffff';
    for (let i = 0; i < 50; i++) {
      const x = (Date.now() / 50 + i * 100) % canvas.width;
      const y = (i * 37) % canvas.height;
      ctx.fillRect(x, y, 1, 1);
    }

    // プレイヤー
    const { player, bullets, enemyBullets, enemies, particles, lives, score, level } = gameState;
    
    // プレイヤー描画
    ctx.fillStyle = '#00ff88';
    ctx.beginPath();
    ctx.moveTo(player.x + player.width / 2, player.y);
    ctx.lineTo(player.x + player.width, player.y + player.height);
    ctx.lineTo(player.x, player.y + player.height);
    ctx.closePath();
    ctx.fill();

    // 弾描画
    ctx.fillStyle = '#ff0';
    bullets.forEach(bullet => {
      ctx.fillRect(bullet.x, bullet.y, bullet.width, bullet.height);
    });

    // 敵弾描画
    ctx.fillStyle = '#f00';
    enemyBullets.forEach(bullet => {
      ctx.fillRect(bullet.x, bullet.y, bullet.width, bullet.height);
    });

    // 敵描画
    enemies.forEach((enemy, i) => {
      const hue = (i * 30 + Date.now() / 100) % 360;
      ctx.fillStyle = `hsl(${hue}, 70%, 60%)`;
      ctx.fillRect(enemy.x, enemy.y, enemy.width, enemy.height);
      
      // 目のような装飾
      ctx.fillStyle = '#000';
      ctx.fillRect(enemy.x + 10, enemy.y + 8, 6, 6);
      ctx.fillRect(enemy.x + enemy.width - 16, enemy.y + 8, 6, 6);
    });

    // パーティクル描画
    particles.forEach(p => {
      ctx.fillStyle = p.color;
      ctx.globalAlpha = p.life / 30;
      ctx.fillRect(p.x, p.y, 3, 3);
      ctx.globalAlpha = 1;
    });

    // UI
    ctx.fillStyle = '#fff';
    ctx.font = '16px monospace';
    ctx.fillText(`SCORE: ${score}`, 20, 30);
    ctx.fillText(`LIVES: ${lives}`, 20, 50);
    ctx.fillText(`LEVEL: ${level}`, 20, 70);

    // ゲームオーバー画面
    if (gameState.isGameOver) {
      ctx.fillStyle = 'rgba(0, 0, 0, 0.7)';
      ctx.fillRect(0, 0, canvas.width, canvas.height);
      
      ctx.fillStyle = '#ff0';
      ctx.font = '48px monospace';
      ctx.textAlign = 'center';
      ctx.fillText('GAME OVER', canvas.width / 2, canvas.height / 2);
      
      ctx.fillStyle = '#fff';
      ctx.font = '20px monospace';
      ctx.fillText(`FINAL SCORE: ${score}`, canvas.width / 2, canvas.height / 2 + 50);
      ctx.fillText('Press SPACE to restart', canvas.width / 2, canvas.height / 2 + 90);
      ctx.textAlign = 'left';
    }
  }, [gameState]);

  useEffect(() => {
    const loop = (timestamp: number) => {
      const deltaTime = timestamp - lastTimeRef.current;
      lastTimeRef.current = timestamp;

      update(deltaTime);
      render();
